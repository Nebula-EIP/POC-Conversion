cmake_minimum_required(VERSION 3.21)
project(nebula_poc VERSION 0.1.0 LANGUAGES CXX)

option(NEBULA_ENABLE_CLANG "Build the Clang adapter when LLVM is found" ON)
option(NEBULA_BUILD_TESTS "Build smoke tests" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(nebula_common STATIC common/nebula.cpp)
target_include_directories(nebula_common PUBLIC common)
add_library(nebula_poc_a STATIC poc_a/templates.cpp)
target_link_libraries(nebula_poc_a PUBLIC nebula_common)
target_include_directories(nebula_poc_a PUBLIC poc_a)
add_library(nebula_poc_c STATIC poc_c/ir.cpp)
target_link_libraries(nebula_poc_c PUBLIC nebula_common nebula_poc_a)
target_include_directories(nebula_poc_c PUBLIC poc_c PRIVATE poc_a)
add_library(nebula_poc_b STATIC poc_b/clang_adapter.cpp)
target_link_libraries(nebula_poc_b PUBLIC nebula_common)
target_include_directories(nebula_poc_b PUBLIC poc_b)

if(NEBULA_ENABLE_CLANG)
  find_package(LLVM CONFIG QUIET)
  if(LLVM_FOUND)
    message(STATUS "LLVM ${LLVM_PACKAGE_VERSION} found")
    target_compile_definitions(nebula_poc_b PRIVATE NEBULA_HAS_CLANG=1)
    target_include_directories(nebula_poc_b PRIVATE ${LLVM_INCLUDE_DIRS})
    target_link_directories(nebula_poc_b PRIVATE ${LLVM_LIBRARY_DIRS})
    target_compile_options(nebula_poc_b PRIVATE ${LLVM_DEFINITIONS})
    if(TARGET clang-cpp)
      target_link_libraries(nebula_poc_b PRIVATE clang-cpp)
    endif()
  else()
    message(WARNING "LLVM/Clang not found: POC B is an explicit unavailable adapter")
  endif()
endif()

add_executable(nebula_harness harness/main.cpp)
target_link_libraries(nebula_harness PRIVATE nebula_poc_a nebula_poc_b nebula_poc_c)
if(NEBULA_BUILD_TESTS)
  enable_testing()
  add_executable(nebula_tests tests/smoke.cpp)
  target_link_libraries(nebula_tests PRIVATE nebula_poc_a nebula_poc_b nebula_poc_c)
  add_test(NAME nebula_smoke COMMAND nebula_tests)
endif()
